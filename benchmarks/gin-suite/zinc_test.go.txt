// Zinc adapter for the gin-gonic/go-http-routing-benchmark suite. Copy it into
// the suite as zinc_test.go and add github.com/0mjs/zinc to its go.mod; the
// upstream files stay untouched. Zinc runs as zinc.New() with its defaults.

package main

import (
	"io"
	"net/http"
	"net/http/httptest"
	"regexp"
	"testing"

	"github.com/0mjs/zinc"
)

func zincHandler(c *zinc.Context) error { return nil }

func zincHandlerWrite(c *zinc.Context) error {
	io.WriteString(c.Writer(), c.Param("name"))
	return nil
}

func zincHandlerTest(c *zinc.Context) error {
	io.WriteString(c.Writer(), c.Request().RequestURI)
	return nil
}

var zincParamRe = regexp.MustCompile(":([^/]*)")

func loadZinc(routes []route) http.Handler {
	h := zincHandler
	if loadTestHandler {
		h = zincHandlerTest
	}
	app := zinc.New()
	for _, r := range routes {
		app.Add(r.method, zincParamRe.ReplaceAllString(r.path, "{$1}"), h)
	}
	return app
}

func loadZincSingle(method, path string, h zinc.HandlerFunc) http.Handler {
	app := zinc.New()
	app.Add(method, zincParamRe.ReplaceAllString(path, "{$1}"), h)
	return app
}

var githubZinc, gplusZinc, parseZinc, staticZinc http.Handler

func init() {
	println("#Zinc routing-structure memory:")
	calcMem("Zinc_GitHub", func() { githubZinc = loadZinc(githubAPI) })
	calcMem("Zinc_GPlus", func() { gplusZinc = loadZinc(gplusAPI) })
	calcMem("Zinc_Parse", func() { parseZinc = loadZinc(parseAPI) })
	calcMem("Zinc_Static", func() { staticZinc = loadZinc(staticRoutes) })
}

func TestZincRouters(t *testing.T) {
	loadTestHandler = true
	defer func() { loadTestHandler = false }()
	for _, api := range apis {
		h := loadZinc(api.routes)
		for _, rt := range api.routes {
			req := httptest.NewRequest(rt.method, rt.path, nil)
			w := httptest.NewRecorder()
			h.ServeHTTP(w, req)
			if w.Code != 200 || w.Body.String() != rt.path {
				t.Errorf("%s %s %s: %d %q", api.name, rt.method, rt.path, w.Code, w.Body.String())
			}
		}
	}
}

// Micro benchmarks

func BenchmarkZinc_Param(b *testing.B) {
	r, _ := http.NewRequest("GET", "/user/gordon", nil)
	benchRequest(b, loadZincSingle("GET", "/user/:name", zincHandler), r)
}

	r, _ := http.NewRequest("GET", path, nil)
	benchRequest(b, h, r)
